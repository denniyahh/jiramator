SCHEMA_VERSION: int = 1
RUNS_DIR: Path = Path(".jiramator/runs")

IssueStatus = Literal["created", "skipped", "failed", "pending"]
RunStatus = Literal["success", "partial", "failed"]
IssueKind = Literal["epic", "per_release", "per_sprint", "imported"]

@dataclass
class IssueResult:
    template_key: str
    kind: IssueKind
    status: IssueStatus
    jira_key: str | None = None
    error: str | None = None

@dataclass
class RunReport:
    schema_version: int = SCHEMA_VERSION
    command: list[str] = field(default_factory=list)
    started_at: str = ""           # ISO8601 UTC
    ended_at: str | None = None
    team_config_path: str = ""     # absolute, resolved
    org_config_path: str = ""      # absolute, resolved
    team_name: str = ""
    pi_label: str | None = None
    versions: list[str] = field(default_factory=list)
    resolved_config_hash: str = ""
    status: RunStatus = "failed"   # pessimistic default — flipped to success only at end
    counts: dict[str, int] = field(default_factory=lambda: {"created": 0, "skipped": 0, "failed": 0})
    issues: list[IssueResult] = field(default_factory=list)

    def to_envelope(self) -> dict[str, Any]: ...
    @classmethod
    def from_envelope(cls, obj: dict[str, Any]) -> "RunReport": ...

class ConfigDriftError(Exception):
    """Raised when --resume detects the resolved-config hash differs from the prior run.

    Caller (cli.py) catches this, formats a user-facing message naming the prior
    and current hashes (first 12 chars), and exits 1 unless --force was passed.
    """

def write_report_atomic(report: RunReport, path: Path) -> None: ...
def default_report_path(team_config_path: Path) -> Path: ...
def find_resumable(team_config_path: Path) -> Path | None: ...
def compute_resolved_hash(
    org_config: "OrgConfig",
    team_config: "TeamConfig",
    pi_label: str | None,
    versions: list[str],
) -> str: ...
```

         payload = {
             "org": org_config.model_dump(mode="json"),
             "team": team_config.model_dump(mode="json"),
             "pi_label": pi_label,
             "versions": list(versions),  # copy — don't trust caller's list
         }
         canonical = json.dumps(payload, sort_keys=True, separators=(",", ":"), ensure_ascii=False)
         return hashlib.sha256(canonical.encode("utf-8")).hexdigest()
         ```
         Key invariants:
         - `sort_keys=True` makes dict ordering irrelevant.
         - `separators=(",", ":")` is a stylistic compactness choice; doesn't affect determinism but reduces the canonical bytes we hash.
         - `ensure_ascii=False` because team names may contain non-ASCII; the `.encode("utf-8")` step still produces bytes.
         - `mode="json"` on `model_dump` ensures Pydantic types like `datetime`/`Path` are serialized as their JSON-compatible forms (strings) — without it, raw types would break `json.dumps`.

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Process → filesystem | Run report bytes cross to disk; partial-write on crash would corrupt resume state. |
| Filesystem → process | On resume, prior report bytes are read back; malformed file should not crash discovery. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-01-10 | Tampering | Crash mid-write leaves half-written report; resume reads partial JSON and behaves unpredictably | mitigate | `tempfile.mkstemp` + `os.fsync` + `os.replace` per write_report_atomic. Verified by Test 8 (force os.replace failure, assert original file intact) |
| T-01-11 | Denial of Service | Corrupt or attacker-planted JSON in `.jiramator/runs/` crashes find_resumable, blocking all resume usage | mitigate | find_resumable wraps per-file load in try/except; logs nothing, continues to next candidate. Verified by Test 17 |
| T-01-12 | Information Disclosure | Run report contains command-line which could include secrets (e.g. `--token sk-...`) if user pipes to argv | accept | `command` field stores `sys.argv` as-is; documented in dataclass docstring "NEVER store env values; argv may contain user-provided flags — caller is responsible for not putting secrets on the CLI." Project's existing convention: token comes from `JIRA_API_TOKEN` env var, never CLI. Plan 04 will reinforce this when populating `command`. No active mitigation in this plan |
| T-01-13 | Tampering | Hash collision lets attacker forge a "drift-not-detected" state | accept | sha256 collision resistance is industry standard; out-of-scope threat for a local-config integrity check |
| T-01-14 | Repudiation | No audit log of which user wrote which report | accept | Single-user CLI tool; OS file ownership is sufficient identity |
</threat_model>
