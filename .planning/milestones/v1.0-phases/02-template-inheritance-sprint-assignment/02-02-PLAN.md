    **Step 1 — Implement `merge_configs` in `jiramator/config_merge.py`:**

    Append:
    ```python
    def merge_configs(
        org_model: "OrgConfig",
        org_tagged_raw: object,
        org_file: Path,
        team_model: "TeamConfig",
        team_tagged_raw: object,
        team_file: Path,
        *,
        console: "Console | None" = None,
    ) -> "TeamConfig":
        """Apply the layered merge: org.default_fields → team.defaults.fields → template fields.

        Mutates ``team_model``: every template's ``fields`` dict is replaced
        with the merged value; ``team_model.defaults.fields`` is replaced with
        the org+team-defaults merged effective layer.

        Emits ConfigConflictWarning entries to ``console`` (or to a fresh
        ``Console(stderr=True)`` if None). Returns the mutated team_model.

        Layer order (CONTEXT working_model):
          1. org.default_fields  (locked org-wide)
          2. team.defaults.fields (locked team-wide)
          3. per-template fields  (fills remaining gaps)
        """
        from jiramator.error_format import ConfigConflictWarning  # noqa: F401 — kept for clarity

        warnings: list[ConfigConflictWarning] = []

        # --- Layer 1: merge org.default_fields INTO team.defaults.fields ---
        org_defaults = org_model.default_fields or {}
        team_defaults_fields = team_model.defaults.fields or {}
        merged_team_defaults, w1 = deep_merge_dicts(
            earlier=org_defaults,
            later=team_defaults_fields,
            path_prefix="defaults.fields",
            earlier_file=org_file,
            later_file=team_file,
            earlier_tagged=org_defaults,
            later_tagged=team_defaults_fields,
            earlier_loc=("default_fields",),
            later_loc=("defaults", "fields"),
            earlier_tagged_root=org_tagged_raw,
            later_tagged_root=team_tagged_raw,
            earlier_layer="org config",
        )
        warnings.extend(w1)
        # Persist the merged effective layer so introspection sees it.
        team_model.defaults.fields = merged_team_defaults

        # --- Layer 2: merge effective team-level locks INTO every template ---
        # Each template needs its own warning context: earlier_layer depends
        # on which key was the locker. Simplification per RESEARCH.md §R4:
        # we just label "team defaults" for layer-2 conflicts; if the user
        # wants to know whether the LOCK came from org or team they can read
        # team_model.defaults.fields (which now holds the merged locks). The
        # layer-1 warnings already attribute org-vs-team conflicts.
        sources = [
            ("recurring_epics",     team_model.recurring_epics),
            ("per_release_tickets", team_model.per_release_tickets),
            ("per_sprint_tickets",  team_model.per_sprint_tickets),
        ]
        for list_name, templates in sources:
            for idx, tmpl in enumerate(templates):
                merged, w_t = deep_merge_dicts(
                    earlier=merged_team_defaults,
                    later=tmpl.fields,
                    path_prefix=f"{list_name}[{idx}].fields",
                    earlier_file=team_file,   # the locks live on the team model now
                    later_file=team_file,
                    earlier_tagged=merged_team_defaults,
                    later_tagged=tmpl.fields,
                    earlier_loc=("defaults", "fields"),
                    later_loc=(list_name, idx, "fields"),
                    earlier_tagged_root=team_tagged_raw,
                    later_tagged_root=team_tagged_raw,
                    earlier_layer="team defaults",
                )
                tmpl.fields = merged
                warnings.extend(w_t)

        # --- Emit warnings ---
        if warnings:
            from rich.console import Console
            stderr = console if console is not None else Console(stderr=True)
            for w in warnings:
                stderr.print(str(w), highlight=False, markup=False)
    ```
